<script>
(() => {
  const grid = document.getElementById('topo-grid');
  const search = document.getElementById('topo-search');
  const countEl = document.getElementById('topo-count');
  const lb = document.getElementById('topo-lightbox');
  const lbImg = document.getElementById('topo-lb-img');
  const lbName = document.getElementById('topo-lb-name');
  const lbElev = document.getElementById('topo-lb-elev');
  const cards = Array.from(grid.querySelectorAll('.topo-card'));
  let visible = cards.slice();
  let curPos = 0;
  function applyFilter() {
    const q = search.value.trim().toLowerCase();
    visible = [];
    for (const c of Array.from(grid.querySelectorAll('.topo-card'))) {
      const match = !q || c.dataset.name.includes(q);
      c.style.display = match ? '' : 'none';
      if (match) visible.push(c);
    }
    countEl.textContent = 'Showing ' + visible.length + ' of ' + cards.length + ' countries';
  }
  function applySort(mode) {
    const sorted = cards.slice().sort((a, b) => {
      if (mode === 'high') return Number(b.dataset.elev) - Number(a.dataset.elev);
      if (mode === 'low') return Number(a.dataset.elev) - Number(b.dataset.elev);
      return a.dataset.admin.localeCompare(b.dataset.admin);
    });
    for (const c of sorted) grid.appendChild(c);
    applyFilter();
  }
  function showAt(pos) {
    if (!visible.length) return;
    curPos = (pos + visible.length) % visible.length;
    const c = visible[curPos];
    lbImg.src = c.dataset.src;
    lbImg.alt = 'Topography of ' + c.dataset.admin;
    lbName.textContent = c.dataset.admin;
    lbElev.textContent = 'max ' + Number(c.dataset.elev).toLocaleString() + ' m';
    lb.classList.add('open');
  }
  search.addEventListener('input', applyFilter);
  for (const btn of document.querySelectorAll('.topo-btn')) {
    btn.addEventListener('click', () => {
      for (const b of document.querySelectorAll('.topo-btn')) b.classList.remove('active');
      btn.classList.add('active');
      applySort(btn.dataset.sort);
    });
  }
  grid.addEventListener('click', (e) => {
    const card = e.target.closest('.topo-card');
    if (!card) return;
    const pos = visible.indexOf(card);
    if (pos !== -1) showAt(pos);
  });
  document.getElementById('topo-prev').addEventListener('click', () => showAt(curPos - 1));
  document.getElementById('topo-next').addEventListener('click', () => showAt(curPos + 1));
  document.getElementById('topo-close').addEventListener('click', () => lb.classList.remove('open'));
  lb.addEventListener('click', (e) => { if (e.target === lb) lb.classList.remove('open'); });
  document.addEventListener('keydown', (e) => {
    if (!lb.classList.contains('open')) return;
    if (e.key === 'Escape') lb.classList.remove('open');
    else if (e.key === 'ArrowLeft') showAt(curPos - 1);
    else if (e.key === 'ArrowRight') showAt(curPos + 1);
  });
})();
